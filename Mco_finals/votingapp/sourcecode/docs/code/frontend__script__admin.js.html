   1  // Page actions for admin.html. Shared wallet and contract logic loads first.
   2  // Render candidate and voter lists and lock setup controls once voting starts.
   3  function showAdmin() {
   4    byId("adminGate").hidden = app.ready && app.admin;
   5    byId("adminContent").hidden = !app.ready || !app.admin;
   6    const setup = app.ready && app.admin && app.data.phase === "setup" && !app.working;
   7    for (const control of document.querySelectorAll("form input, form button")) control.disabled = !setup;
   8    byId("refreshButton").disabled = !app.configured || app.working;
   9    byId("startButton").disabled = !setup || app.data.candidates.length < 2 || app.data.approved < 1;
  10    byId("closeButton").disabled = !app.ready || !app.admin || app.working || app.data.phase !== "open";
  11    if (!app.ready || !app.admin) return;
  12    byId("candidateCount").textContent = app.data.candidates.length + " added";
  13    byId("voterCount").textContent = app.data.approved + " approved";
  14    byId("phase").textContent = app.data.phase === "setup" ? "Setup" : app.data.phase === "open" ? "Open" : "Closed";
  15    if (app.data.position) byId("candidatePosition").value = app.data.position;
  16    byId("candidatePosition").readOnly = !!app.data.position;
  17    byId("adminCandidates").replaceChildren();
  18    for (const candidate of app.data.candidates) {
  19      const row = app.copy("adminCandidateRow");
  20      row.querySelector(".candidate-name").textContent = candidate.name;
  21      row.querySelector(".candidate-position").textContent = candidate.position;
  22      byId("adminCandidates").append(row);
  23    }
  24    byId("registeredVoters").replaceChildren();
  25    byId("emptyVoters").hidden = app.data.voters.length > 0;
  26    for (const voter of app.data.voters) {
  27      const row = app.copy("voterRow");
  28      row.querySelector(".wallet-address").textContent = voter.address;
  29      row.querySelector(".voter-status").textContent = voter.approved ? "Authorized" : "Awaiting approval";
  30      const button = row.querySelector("button");
  31      button.hidden = voter.approved || app.data.phase !== "setup";
  32      button.disabled = app.working;
  33      button.onclick = function () { byId("voterAddress").value = voter.address; byId("voterAddress").focus(); };
  34      byId("registeredVoters").append(row);
  35    }
  36  }
  37  byId("candidateForm").onsubmit = function (event) {
  38    event.preventDefault();
  39    if (byId("addCandidateButton").disabled) return;
  40    const name = byId("candidateName").value.trim().replace(/\s+/g, " ");
  41    const position = byId("candidatePosition").value.trim().replace(/\s+/g, " ");
  42    if (!name || !position) { app.note("Enter a name and position.", true); return; }
  43    if (ethers.utils.toUtf8Bytes(name).length > 120 || ethers.utils.toUtf8Bytes(position).length > 120) { app.note("Use shorter text.", true); return; }
  44    if (app.data.candidates.some(item => app.same(item.name, name))) { app.note("Candidate already added.", true); return; }
  45    app.confirm("Add this candidate?", name + " · " + position + ". Check spelling; this cannot be edited.", function () {
  46      app.send("addCandidate", [name, position], "Candidate added.", function () { byId("candidateName").value = ""; });
  47    });
  48  };
  49  byId("authorizeForm").onsubmit = function (event) {
  50    event.preventDefault();
  51    if (byId("authorizeButton").disabled) return;
  52    const address = byId("voterAddress").value.trim();
  53    if (!ethers.utils.isAddress(address) || address === ethers.constants.AddressZero) { app.note("Enter a valid voter wallet address.", true); return; }
  54    if (app.same(address, app.account)) { app.note("Use a separate voter wallet.", true); return; }
  55    if (app.data.voters.some(item => app.same(item.address, address) && item.approved)) { app.note("Wallet already authorized.", true); return; }
  56    app.confirm("Authorize this voter?", address + ". Verify one wallet per student.", function () {
  57      app.send("authorizeVoter", [address], "Voter authorized.", function () { byId("voterAddress").value = ""; });
  58    });
  59  };
  60  byId("startButton").onclick = function () {
  61    if (!byId("startButton").disabled) app.confirm("Start voting?", "Candidate and voter setup will be locked.", function () {
  62      app.send("startElection", [], "Voting is now open.");
  63    });
  64  };
  65  byId("closeButton").onclick = function () {
  66    if (!byId("closeButton").disabled) app.confirm("Close voting permanently?", "Voting cannot be reopened.", function () {
  67      app.send("endElection", [], "Voting is closed.");
  68    });
  69  };
  70  byId("refreshButton").onclick = app.read;
  71  app.start(showAdmin);
