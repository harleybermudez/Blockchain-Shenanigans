   1  // Page actions for vote.html. Shared wallet and contract logic loads first.
   2  let selectedCandidate = 0;
   3  // Keep the ballot summary and review button aligned with the selected candidate.
   4  function updateSelection() {
   5    const candidate = app.data.candidates.find(item => item.id === selectedCandidate);
   6    byId("selectionName").textContent = candidate ? candidate.name : "No candidate selected";
   7    byId("selectionPosition").textContent = app.data.position;
   8    byId("reviewButton").disabled = app.working || !app.ready || !app.allowed || app.voted || !candidate || app.data.phase !== "open";
   9    for (const radio of byId("candidateChoices").querySelectorAll("input")) radio.disabled = app.working;
  10  }
  11  byId("ballotForm").onsubmit = function (event) {
  12    event.preventDefault();
  13    if (byId("reviewButton").disabled) return;
  14    const candidate = app.data.candidates.find(item => item.id === selectedCandidate);
  15    app.confirm("Review your vote", candidate.name + " · " + candidate.position + ". A confirmed vote cannot be changed.", function () {
  16      app.send("vote", [candidate.id], "Your vote is recorded.", function () { selectedCandidate = 0; });
  17    });
  18  };
  19  app.start(function () {
  20    const eligible = app.ready && app.account && !app.admin && app.allowed && !app.voted && app.data.phase === "open";
  21    const recorded = app.ready && app.voted && !app.admin;
  22    byId("ballotForm").hidden = !eligible;
  23    byId("selectionPanel").hidden = !eligible;
  24    byId("voteGate").hidden = !!eligible || recorded;
  25    byId("voteSuccess").hidden = !recorded;
  26    byId("ballotLayout").classList.toggle("is-blocked", !eligible);
  27    byId("position").textContent = app.data.position || "Select one candidate.";
  28    byId("eligibility").textContent = recorded ? "Vote recorded" : eligible ? "Eligible to vote" : "";
  29    let status = ["Voting is not available yet", "View candidates", "candidates.html"];
  30    if (app.ready) {
  31      if (app.data.phase === "closed") status = ["Voting has closed", "View results", "results.html"];
  32      else if (!app.account) status = ["Log in to cast your vote", "Voter login", "index.html"];
  33      else if (app.admin) status = ["Use a separate voter wallet", "Manage election", "admin.html"];
  34      else if (!app.registered) status = ["Register before voting", "Sign up", "signup.html"];
  35      else if (!app.allowed) status = ["Waiting for admin approval", "View candidates", "candidates.html"];
  36      else if (app.data.phase === "setup") status = ["You are ready. Wait for voting to open.", "View candidates", "candidates.html"];
  37    }
  38    byId("gateTitle").textContent = status[0];
  39    byId("gateLink").textContent = status[1];
  40    byId("gateLink").href = status[2];
  41    if (!eligible) { selectedCandidate = 0; byId("candidateChoices").replaceChildren(); }
  42    if (eligible && !byId("candidateChoices").children.length) {
  43      for (const candidate of app.data.candidates) {
  44        const option = app.copy("candidateOption");
  45        option.querySelector(".candidate-name").textContent = candidate.name;
  46        option.querySelector(".candidate-position").textContent = candidate.position;
  47        const radio = option.querySelector("input");
  48        radio.value = candidate.id;
  49        radio.onchange = function () { selectedCandidate = candidate.id; updateSelection(); };
  50        byId("candidateChoices").append(option);
  51      }
  52    }
  53    updateSelection();
  54  });
