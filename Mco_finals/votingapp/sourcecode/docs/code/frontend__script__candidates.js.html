   1  // Page actions for candidates.html. Shared wallet and contract logic loads first.
   2  app.start(function () {
   3    byId("position").textContent = app.data.position;
   4    byId("emptyCandidates").hidden = app.data.candidates.length > 0;
   5    byId("candidateList").replaceChildren();
   6    for (const candidate of app.data.candidates) {
   7      const row = app.copy("candidateRow");
   8      row.querySelector(".initials").textContent = candidate.name.trim().split(/\s+/).slice(0, 2).map(word => word[0] || "").join("").toUpperCase();
   9      row.querySelector(".candidate-name").textContent = candidate.name;
  10      row.querySelector(".candidate-position").textContent = candidate.position;
  11      row.querySelector(".candidate-votes").textContent = candidate.votes + " votes";
  12      row.querySelector(".candidate-votes").hidden = !app.ready || app.data.phase !== "closed";
  13      byId("candidateList").append(row);
  14    }
  15  });
