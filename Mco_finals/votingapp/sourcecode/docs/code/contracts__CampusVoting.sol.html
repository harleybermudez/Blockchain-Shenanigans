   1  // SPDX-License-Identifier: MIT
   2  pragma solidity ^0.8.24;
   3  
   4  import "@openzeppelin/contracts@5.0.2/access/Ownable.sol";
   5  
   6  // One deployed contract is one election for one position.
   7  contract CampusVoting is Ownable {
   8      struct Candidate {
   9          uint256 id;
  10          string name;
  11          string position;
  12          uint256 voteCount;
  13          bool exists;
  14      }
  15  
  16      uint256 public candidateCount;
  17      uint256 public totalVotes;
  18      uint256 public authorizedVoterCount;
  19      bool public electionOpen;
  20      bool public electionClosed;
  21      string public electionPosition;
  22  
  23      // These mappings are the authoritative registration, eligibility, and vote records.
  24      mapping(uint256 => Candidate) public candidates;
  25      mapping(address => bool) public registeredVoters;
  26      mapping(address => bool) public authorizedVoters;
  27      mapping(address => bool) public hasVoted;
  28      address[] private voterAddresses;
  29  
  30      event VoterRegistered(address indexed voter);
  31      event CandidateAdded(uint256 indexed candidateId, string name);
  32      event VoterAuthorized(address indexed voter);
  33      event VoteCast(address indexed voter, uint256 indexed candidateId);
  34      event ElectionStatusChanged(bool isOpen);
  35  
  36      constructor() Ownable(msg.sender) {}
  37  
  38      modifier duringSetup() {
  39          require(!electionOpen && !electionClosed, "Election setup has ended");
  40          _;
  41      }
  42  
  43      // Record the caller as registered during setup. Admin approval is a separate step.
  44      function registerVoter() external duringSetup {
  45          require(msg.sender != owner(), "Use a separate voter wallet");
  46          require(!registeredVoters[msg.sender], "Wallet is already registered");
  47          registeredVoters[msg.sender] = true;
  48          voterAddresses.push(msg.sender);
  49          emit VoterRegistered(msg.sender);
  50      }
  51  
  52      // Only the owner may add candidates. The first candidate fixes the election position.
  53      function addCandidate(string calldata _name, string calldata _position)
  54          external onlyOwner duringSetup
  55      {
  56          require(bytes(_name).length > 0 && bytes(_name).length <= 120, "Enter a candidate name (max 120 bytes)");
  57          require(bytes(_position).length > 0 && bytes(_position).length <= 120, "Enter a position (max 120 bytes)");
  58          if (candidateCount == 0) {
  59              electionPosition = _position;
  60          } else {
  61              require(keccak256(bytes(_position)) == keccak256(bytes(electionPosition)), "Use the same election position");
  62          }
  63          for (uint256 i = 1; i <= candidateCount; i++) {
  64              require(keccak256(bytes(candidates[i].name)) != keccak256(bytes(_name)), "Candidate already exists");
  65          }
  66          candidateCount++;
  67          candidates[candidateCount] = Candidate(candidateCount, _name, _position, 0, true);
  68          emit CandidateAdded(candidateCount, _name);
  69      }
  70  
  71      // Approve a verified wallet. An address may also be registered directly by the owner.
  72      function authorizeVoter(address _voter) external onlyOwner duringSetup {
  73          require(_voter != address(0), "Invalid voter address");
  74          require(_voter != owner(), "Use a separate voter wallet");
  75          require(!authorizedVoters[_voter], "Voter is already authorized");
  76          // The admin may also register a verified address directly.
  77          if (!registeredVoters[_voter]) {
  78              registeredVoters[_voter] = true;
  79              voterAddresses.push(_voter);
  80              emit VoterRegistered(_voter);
  81          }
  82          authorizedVoters[_voter] = true;
  83          authorizedVoterCount++;
  84          emit VoterAuthorized(_voter);
  85      }
  86  
  87      // Require candidates and an approved voter, then permanently lock election setup.
  88      function startElection() external onlyOwner duringSetup {
  89          require(candidateCount >= 2, "Add at least two candidates");
  90          require(authorizedVoterCount > 0, "Authorize at least one voter");
  91          electionOpen = true;
  92          emit ElectionStatusChanged(true);
  93      }
  94  
  95      // Close voting permanently. The duringSetup modifier also prevents reopening.
  96      function endElection() external onlyOwner {
  97          require(electionOpen, "Election is not open");
  98          electionOpen = false;
  99          electionClosed = true;
 100          emit ElectionStatusChanged(false);
 101      }
 102  
 103      // Enforce eligibility and one vote per wallet on chain, then increment both vote counters.
 104      function vote(uint256 _candidateId) external {
 105          require(electionOpen, "Election is not open");
 106          require(authorizedVoters[msg.sender], "Your wallet is not authorized");
 107          require(!hasVoted[msg.sender], "Your vote is already recorded");
 108          require(candidates[_candidateId].exists, "Invalid candidate");
 109          hasVoted[msg.sender] = true;
 110          candidates[_candidateId].voteCount++;
 111          totalVotes++;
 112          emit VoteCast(msg.sender, _candidateId);
 113      }
 114  
 115      // Return a candidate only when its ID exists.
 116      function getCandidate(uint256 _candidateId) external view returns (Candidate memory) {
 117          require(candidates[_candidateId].exists, "Invalid candidate");
 118          return candidates[_candidateId];
 119      }
 120  
 121      function getCandidateCount() external view returns (uint256) { return candidateCount; }
 122      function getTotalVotes() external view returns (uint256) { return totalVotes; }
 123      function getElectionStatus() external view returns (bool) { return electionOpen; }
 124      function getRegisteredVoters() external view returns (address[] memory) { return voterAddresses; }
 125  }
