   2  async function walletMatches(wallet) {
   3    const accounts = await window.ethereum.request({ method: "eth_accounts" });
   4    const chain = await window.ethereum.request({ method: "eth_chainId" });
   5    return accounts.length > 0 && app.same(wallet, accounts[0]) && Number(chain) === settings.chainId;
   6  }
   7  // The app object holds shared state and methods. HTML templates define the rows.
   8  function byId(id) { return document.getElementById(id); }
   9  const app = {
  10    account: "", admin: false, registered: false, allowed: false, voted: false,
  11    configured: false, ready: false, working: false, revision: 0,
  12    reader: null, contract: null, update: function () {},
  13    data: { phase: "setup", candidates: [], voters: [], position: "", total: 0, approved: 0 },
  14    session: function () { return "voting-v2:" + settings.address; },
  15    same: function (a, b) { return a.toLowerCase() === b.toLowerCase(); },
  16    copy: function (id) { return byId(id).content.firstElementChild.cloneNode(true); },
  17    note: function (text, error = false) {
  18      byId("notice").hidden = false;
  19      byId("notice").classList.toggle("is-error", error);
  20      byId("notice").setAttribute("role", error ? "alert" : "status");
  21      byId("notice").textContent = text;
  22    },
  23    draw: function () {
  24      byId("electionStatus").hidden = !app.configured;
  25      byId("electionStatus").textContent = !app.ready ? "Election unavailable" :
  26        app.data.phase === "open" ? "Voting open" : app.data.phase === "closed" ? "Voting closed" : "Voting not started";
  27      byId("electionStatus").dataset.state = app.data.phase;
  28      byId("manageLink").hidden = !app.admin;
  29      byId("logoutButton").hidden = !app.account;
  30      byId("loginLink").hidden = !!app.account;
  31      app.update();
  32    },
  33    // Clear role and eligibility data and invalidate any outstanding read.
  34    reset: function () {
  35      app.revision++;
  36      sessionStorage.removeItem(app.session());
  37      app.account = "";
  38      app.admin = app.registered = app.allowed = app.voted = false;
  39      app.data.voters = [];
  40      if (byId("reviewDialog") && byId("reviewDialog").open) byId("reviewDialog").close();
  41      app.draw();
  42    },
  43    // Restore a saved wallet only when MetaMask still exposes it on the configured chain.
  44    restore: async function () {
  45      const saved = sessionStorage.getItem(app.session());
  46      if (!saved || !window.ethereum) return;
  47      if (await walletMatches(saved)) app.account = saved;
  48      else sessionStorage.removeItem(app.session());
  49    },
  50    // Read a single block and ignore stale responses after an account or network change.
  51    read: async function () {
  52      if (!app.configured || !app.contract) return;
  53      // A newer request or wallet change makes this response obsolete.
  54      const revision = ++app.revision;
  55      const wallet = app.account;
  56      try {
  57        const block = { blockTag: await app.reader.getBlockNumber() };
  58        const info = await Promise.all([
  59          app.contract.owner(block), app.contract.getElectionStatus(block), app.contract.electionClosed(block),
  60          app.contract.getCandidateCount(block), app.contract.getTotalVotes(block),
  61          app.contract.electionPosition(block), app.contract.authorizedVoterCount(block)
  62        ]);
  63        const candidates = [], voters = [];
  64        for (let id = 1; id <= Number(info[3]); id++) {
  65          const c = await app.contract.getCandidate(id, block);
  66          candidates.push({ id: Number(c.id), name: c.name, position: c.position, votes: Number(c.voteCount) });
  67        }
  68        const admin = !!wallet && app.same(wallet, info[0]);
  69        let flags = [false, false, false];
  70        if (wallet) flags = await Promise.all([
  71          app.contract.registeredVoters(wallet, block), app.contract.authorizedVoters(wallet, block), app.contract.hasVoted(wallet, block)
  72        ]);
  73        if (admin && document.body.dataset.page === "admin") {
  74          for (const address of await app.contract.getRegisteredVoters(block)) {
  75            voters.push({ address: address, approved: await app.contract.authorizedVoters(address, block) });
  76          }
  77        }
  78        if (revision !== app.revision || wallet !== app.account) return;
  79        app.data = { phase: info[2] ? "closed" : info[1] ? "open" : "setup", candidates: candidates,
  80          voters: voters, total: Number(info[4]), position: info[5], approved: Number(info[6]) };
  81        app.admin = admin;
  82        [app.registered, app.allowed, app.voted] = flags;
  83        app.ready = true;
  84        app.draw();
  85      } catch {
  86        if (revision !== app.revision || wallet !== app.account) return;
  87        app.ready = false;
  88        app.draw();
  89        app.note("Could not load the election. Refresh to try again.", true);
  90      }
  91    },
  92    // Connect MetaMask, verify the requested role, and route the voter or administrator.
  93    connect: async function (role) {
  94      if (app.working || !app.configured) return;
  95      app.working = true;
  96      app.draw();
  97      try {
  98        if (!window.ethereum) { app.note("Install MetaMask to continue.", true); return; }
  99        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
 100        const chain = await window.ethereum.request({ method: "eth_chainId" });
 101        if (Number(chain) !== settings.chainId) { app.note("Select " + settings.network + " in MetaMask.", true); return; }
 102        const owner = await app.contract.owner();
 103        const ownerWallet = app.same(accounts[0], owner);
 104        if ((role === "admin") !== ownerWallet) {
 105          app.note(ownerWallet ? "Use electoral admin login." : "Select the admin wallet.", true);
 106          return;
 107        }
 108        app.account = accounts[0];
 109        sessionStorage.setItem(app.session(), app.account);
 110        await app.read();
 111        if (!app.ready) return;
 112        if (role === "signup") app.note(app.registered ? "Wallet already registered." : "Wallet connected.");
 113        else window.location.href = role === "admin" ? "admin.html" : app.registered ? "vote.html" : "signup.html";
 114      } catch { app.note("Could not connect. Check MetaMask and try again.", true); }
 115      finally { app.working = false; app.draw(); }
 116    },
 117    // Recheck wallet and chain, then wait for a successful transaction receipt.
 118    send: async function (method, args, success, after) {
 119      if (app.working || !app.ready || !app.account) return;
 120      const wallet = app.account;
 121      app.working = true;
 122      app.draw();
 123      try {
 124        if (!await walletMatches(wallet)) {
 125          app.reset(); app.note("Wallet changed. Log in again.", true); return;
 126        }
 127        const provider = new ethers.providers.Web3Provider(window.ethereum, "any");
 128        const writer = app.contract.connect(provider.getSigner(wallet));
 129        app.note("Confirm in MetaMask.");
 130        const transaction = await writer[method](...args);
 131        app.note("Recording… Please wait.");
 132        const receipt = await transaction.wait();
 133        if (receipt.status !== 1) throw new Error();
 134        await app.read();
 135        app.note(wallet === app.account ? success : "Transaction confirmed. Log in again.");
 136        if (wallet === app.account && after) after();
 137      } catch {
 138        app.note("Action could not be confirmed. Refresh before trying again.", true);
 139        await app.read();
 140      } finally { app.working = false; app.draw(); }
 141    },
 142    // Ask the user to review an action before sending its transaction.
 143    confirm: function (title, description, action) {
 144      if (app.working) return;
 145      byId("reviewTitle").textContent = title;
 146      byId("reviewText").textContent = description;
 147      byId("confirmAction").onclick = function () { byId("reviewDialog").close(); action(); };
 148      byId("reviewDialog").showModal();
 149      byId("cancelAction").focus();
 150    },
 151    // Load frontend/abi.json and start the page after its render callback is registered.
 152    start: async function (update) {
 153      app.update = update;
 154      app.draw();
 155      if (!window.ethers) { app.note("Wallet library unavailable. Reload the page.", true); return; }
 156      app.configured = ethers.utils.isAddress(settings.address) && settings.address !== ethers.constants.AddressZero;
 157      if (app.configured) {
 158        app.reader = new ethers.providers.JsonRpcProvider(settings.rpc, settings.chainId);
 159        try {
 160          // Use the compiler-generated interface for both reads and signed calls.
 161        const response = await fetch("../abi.json");
 162          if (!response.ok) throw new Error();
 163          app.contract = new ethers.Contract(settings.address, await response.json(), app.reader);
 164        } catch { app.configured = false; app.draw(); app.note("Could not load the contract interface. Reload the page.", true); return; }
 165        try { await app.restore(); } catch { sessionStorage.removeItem(app.session()); }
 166        await app.read();
 167      }
 168      app.draw();
 169    }
 170  };
