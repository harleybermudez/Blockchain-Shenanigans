   1  // Page actions for results.html. Shared wallet and contract logic loads first.
   2  byId("refreshButton").onclick = app.read;
   3  app.start(function () {

  10    byId("totalVotes").textContent = app.data.total;

  14      row.querySelector(".candidate-name").textContent = candidate.name;
  15      row.querySelector(".candidate-count").textContent = candidate.votes;
  16      byId("resultRows").append(row);
  17    }
  18  });
